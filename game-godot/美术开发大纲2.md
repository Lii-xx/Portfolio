    var p = GPUParticles2D.new()

func _load_background(bg_name: String) -> void:
    var path := "res://assets/backgrounds/"
    var full_path: String
    if bg_name != "" and ResourceLoader.exists(path + bg_name + ".png"):
        full_path = path + bg_name + ".png"

### 4.3 实现要点

**`tween_helpers.gd` 新增函数**：
```gdscript
## 奖励卡牌弹出
static func reward_card_popup(card: Control) -> void:
    card.scale = Vector2(0, 0)
    var tween = card.create_tween()
    tween.tween_property(card, "scale", Vector2(1.1, 1.1), 0.3).set_ease(Tween.EASE_OUT)
    tween.tween_property(card, "scale", Vector2(1.0, 1.0), 0.1)

## 怪物意图切换
static func intent_switch(label: Label) -> void:
    var tween = label.create_tween()
    tween.tween_property(label, "modulate:a", 0.0, 0.15)
    tween.tween_property(label, "modulate:a", 1.0, 0.15)

## 火堆火焰跳动（循环）
static func campfire_flicker(node: Control) -> void:
    var tween = node.create_tween().set_loops()
    tween.tween_property(node, "modulate", Color(1.2, 1.0, 0.8, 1), 0.3)
    tween.tween_property(node, "modulate", Color(1.0, 1.0, 1.0, 1), 0.3)
    tween.tween_property(node, "modulate", Color(0.9, 1.1, 1.2, 1), 0.3)
    tween.tween_property(node, "modulate", Color(1.0, 1.0, 1.0, 1), 0.3)

## 怪物待机呼吸（循环）
static func monster_idle_breath(node: Node2D) -> void:
    var orig_scale = node.scale
    var tween = node.create_tween().set_loops()
    tween.tween_property(node, "scale", orig_scale * 1.03, 1.2).set_ease(Tween.EASE_IN_OUT)
    tween.tween_property(node, "scale", orig_scale * 0.97, 1.2).set_ease(Tween.EASE_IN_OUT)

## BOSS 登场
static func boss_entrance(boss_node: Node2D) -> void:
    var target_pos = boss_node.position
    boss_node.position = target_pos + Vector2(0, -400)
    boss_node.scale = Vector2(0.1, 0.1)
    boss_node.modulate.a = 0.0
    var tween = boss_node.create_tween()
    tween.set_parallel(true)
    tween.tween_property(boss_node, "position", target_pos, 0.8).set_ease(Tween.EASE_OUT)
    tween.tween_property(boss_node, "scale", Vector2(1.0, 1.0), 0.8).set_ease(Tween.EASE_OUT)
    tween.tween_property(boss_node, "modulate:a", 1.0, 0.5)
```

### 4.4 接入点
- `reward_modal.gd`：卡牌创建后调 `reward_card_popup`
- `monster_slot.gd`：意图刷新时调 `intent_switch`；`_ready` 时启动 `monster_idle_breath`（注意：死亡时停止呼吸，避免与 dissolve 冲突）
- `campfire_screen.gd`：`_ready` 时对火焰节点调 `campfire_flicker`
- `battle_screen.gd`：BOSS 关刷新怪物时，对 BOSS 调 `boss_entrance`

### 4.5 注意事项
- **循环 Tween 必须能被停止**：怪物死亡时 `kill` 待机呼吸 Tween，避免与 dissolve 冲突
- BOSS 登场动画期间应禁用玩家输入，避免动画被打断
- 待机呼吸的 scale 变化幅度要小（±3%），过大显得滑稽

### 4.6 验证
- `run_project` 无报错
- 各氛围动效可见
